cmake_minimum_required(VERSION 3.20)
project(lamsa_store VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_library(sqlite STATIC vendor/sqlite3.c)
target_include_directories(sqlite PUBLIC vendor)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_FOREIGN_KEYS=1)
target_link_libraries(sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_executable(lamsa backend/main.cpp backend/images.cpp)
target_include_directories(lamsa PRIVATE vendor backend)
target_link_libraries(lamsa PRIVATE sqlite OpenSSL::Crypto Threads::Threads)
if(WIN32)
 target_link_libraries(lamsa PRIVATE ws2_32 crypt32)
 target_compile_definitions(lamsa PRIVATE _WIN32_WINNT=0x0A00)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(lamsa PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_FOUND)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:lamsa> ${CMAKE_SOURCE_DIR})
 endif()
endif()
